import { Shader, Swirl, ChromaFlow } from "shaders/react";

export function BrandShader() {
  return (
    <Shader disableTelemetry>
      <Swirl blend={50} detail={1.7}
        stops={["#e6e2ff", "#bdb2ff"]} />
      <ChromaFlow baseColor="#ae00ff"
        downColor="#4642ff" leftColor="#5900ff"
        rightColor="#5b4fff" upColor="#7105ff" />
    </Shader>
  );
}
